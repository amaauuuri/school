
// ========================================================
// PRACTICA 1: CONTROL DE MOTORES DESDE MONITOR SERIAL
// ========================================================

const int IN1 = 16;
const int IN2 = 17;
const int IN3 = 18;
const int IN4 = 19;

void setup() {
  Serial.begin(115200);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);

  detener();

  Serial.println("==========================================");
  Serial.println("  CONTROL DE CARRITO CON ESP32 Y L298N    ");
  Serial.println("==========================================");
  Serial.println("Ingresa un numero para mover el carrito:");
  Serial.println(" 1 -> Adelante");
  Serial.println(" 2 -> Atras");
  Serial.println(" 3 -> Giro a la Derecha");
  Serial.println(" 4 -> Giro a la Izquierda");
  Serial.println(" 5 -> Paro Total");
  Serial.println("------------------------------------------");
}

void loop() {
  if (Serial.available() > 0) {
    char boton = Serial.read();

    switch (boton) {
      case '1':
        adelante();
        Serial.println("Estatus: [ 1 ] MOVIENDO ADELANTE");
        break;

      case '2':
        atras();
        Serial.println("Estatus: [ 2 ] MOVIENDO ATRAS");
        break;

      case '3':
        derecha();
        Serial.println("Estatus: [ 3 ] GIRANDO A LA DERECHA");
        break;

      case '4':
        izquierda();
        Serial.println("Estatus: [ 4 ] GIRANDO A LA IZQUIERDA");
        break;

      case '5':
        detener();
        Serial.println("Estatus: [ 5 ] PARO TOTAL");
        break;
    }
  }
}

// ========================================================
// FUNCIONES DE CONTROL DE MOVIMIENTO
// ========================================================

void adelante() {
  // Motor Izquierdo avanza
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void atras() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void derecha() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}

void izquierda() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void detener() {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}
