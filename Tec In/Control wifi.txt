#include <WiFi.h>
#include <WebServer.h>

const char *ssid = "Carrito_WiFi";
const char *password = "12345678";

WebServer server(80);

const int IN1 = 16;
const int IN2 = 17;
const int IN3 = 18;
const int IN4 = 19;

// Página HTML del control remoto
const char HTTP_PAGE[] PROGMEM = R"rawliteral(
<!DOCTYPE html>
<html>
<head>
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <style>
    body { font-family: Arial; text-align: center; background-color: #222; color: white; margin:0; padding:20px; }
    h1 { margin-bottom: 20px; }
    .btn { background-color: #008CBA; border: none; color: white; padding: 20px 30px; text-align: center; text-decoration: none; display: inline-block; font-size: 18px; margin: 8px; border-radius: 12px; touch-action: manipulation; width: 120px; }
    .stop { background-color: #f44336; width: 260px; }
    .row { margin: 10px 0; }
  </style>
</head>
<body>
  <h1>Control Carrito Wi-Fi</h1>
  <div class="row"><button class="btn" onclick="sendCmd('/1')">AVANZAR</button></div>
  <div class="row">
    <button class="btn" onclick="sendCmd('/4')">IZQUIERDA</button>
    <button class="btn" onclick="sendCmd('/3')">DERECHA</button>
  </div>
  <div class="row"><button class="btn" onclick="sendCmd('/2')">REVERSA</button></div>
  <div class="row"><button class="btn stop" onclick="sendCmd('/5')">STOP / PARO</button></div>

  <script>
    function sendCmd(cmd) {
      fetch(cmd);
    }
  </script>
</body>
</html>
)rawliteral";

void handleRoot() {
  server.send(200, "text/html", HTTP_PAGE);
}

void setup() {
  Serial.begin(115200);

  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);
  detener();

  WiFi.softAP(ssid, password);
  IPAddress IP = WiFi.softAPIP();
  Serial.print("Red Wi-Fi creada: ");
  Serial.println(ssid);
  Serial.print("Direccion IP para navegar: ");
  Serial.println(IP);

  server.on("/", handleRoot);
  server.on("/1", []() { adelante(); server.send(200, "text/plain", "OK"); });
  server.on("/2", []() { atras(); server.send(200, "text/plain", "OK"); });
  server.on("/3", []() { derecha(); server.send(200, "text/plain", "OK"); });
  server.on("/4", []() { izquierda(); server.send(200, "text/plain", "OK"); });
  server.on("/5", []() { detener(); server.send(200, "text/plain", "OK"); });

  server.begin();
}

void loop() {
  server.handleClient();
}

void adelante() {
  digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);  digitalWrite(IN4, HIGH);
}

void atras() {
  digitalWrite(IN1, LOW);  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW);
}

void derecha() {
  digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);  digitalWrite(IN4, LOW);
}

void izquierda() {
  digitalWrite(IN1, LOW);  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);  digitalWrite(IN4, HIGH);
}

void detener() {
  digitalWrite(IN1, LOW);  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);  digitalWrite(IN4, LOW);
}
